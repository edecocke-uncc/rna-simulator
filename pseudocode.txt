get_gc_content(sequence):
    raise error if empty or invalid characters
    count all G and C nucleotides
    return (gc_count / total_length) * 100

get_ambiguity_content(sequence):
    raise error if empty
    count all characters in AMBIGUOUS set
    return (amb_count / total_length) * 100

generate_random_codon():
    pick 3 random nucleotides from VALID_RNA
    return joined string

is_start_codon(codon):
    raise error if length != 3
    return True if codon == AUG, else False

is_stop_codon(codon):
    raise error if length != 3
    return True if codon in STOP_CODONS, else False

generate_random_sequence(length, ambiguity_rate=0.0):
    raise error if length <= 0
    for each position:
        if random draw < ambiguity_rate: pick from AMBIGUOUS
        else: pick from VALID_RNA
    return joined sequence string

write_fasta(sequences, output_file):
    raise error if list is empty
    open file for writing
    for each (id, description, sequence):
        write ">id description"
        write sequence on next line

__init__(all parameters):
    validate all inputs, raise ValueError if out of range
    store all as instance variables

generate_orf(complete=True):
    pick random codon_count between min and max orf length
    if complete:
        build AUG + (codon_count - 2) random codons + random stop codon
        return full sequence
    if partial:
        return random sequence of (codon_count * 3) nucleotides

generate_sequence():
    decide if ORF is complete using completeness_ratio
    generate ORF via generate_orf()
    decide if flanked using flanking_probability
    if flanked:
        generate left and right sequences via generate_random_sequence()
        return left + orf + right
    else: return orf

generate_sequences():
    for i in 1 to num_sequences:
        call generate_sequence()
        calculate gc via get_gc_content()
        calculate ambiguity via get_ambiguity_content()
        type = "complete" if sequence starts with AUG, else "partial"

save_fasta(output_file):
    call generate_sequences()

create_parser():
    define argparse arguments with defaults:
        num_sequences=10, output="sequences.fasta",
        min_length=100, max_length=1000,
        flanking_prob=0.5, flanking_length=50, completeness=0.7
    return configured parser

validate_arguments(args):
    raise ValueError if min_length > max_length
    raise ValueError if any probability is outside 0-1
    raise ValueError if num_sequences <= 0

main():
    parse arguments via create_parser()
    validate via validate_arguments()
    create Simulator instance with parsed arguments
    call save_fasta() with output path
    print success message to stdout
    return 0 on success, 1 on error
